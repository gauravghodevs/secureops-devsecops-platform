from fastapi import FastAPI, Response
from prometheus_client import (
    CONTENT_TYPE_LATEST,
    REGISTRY,
    Counter,
    generate_latest,
)

app = FastAPI(title="SecureOps Payment API", version="0.1.0")

REQUEST_COUNT = Counter(
    "payment_api_requests_total",
    "Total number of HTTP requests handled by the payment API.",
)

HTTP_RESPONSES = Counter(
    "payment_api_responses_total",
    "Total number of HTTP responses returned by the payment API.",
    ["status_code"],
)


@app.middleware("http")
async def count_requests(request, call_next):
    # Prometheus polling is telemetry collection, not application traffic.
    if request.url.path.rstrip("/") == "/metrics":
        return await call_next(request)

    REQUEST_COUNT.inc()
    response = await call_next(request)
    HTTP_RESPONSES.labels(status_code=str(response.status_code)).inc()
    return response


@app.get("/")
def root():
    return {"service": "payment-api", "status": "running"}


@app.get("/health")
def health():
    return {"status": "healthy"}


@app.get("/ready")
def ready():
    return {"status": "ready"}


@app.get("/metrics", include_in_schema=False)
def metrics():
    return Response(
        content=generate_latest(REGISTRY),
        headers={"Content-Type": CONTENT_TYPE_LATEST},
    )

"""Configuration-based selection of secret providers."""

import os

from app.secrets import (
    AwsSecretsManagerProvider,
    LocalMockSecretProvider,
    SecretProvider,
)


def create_secret_provider() -> SecretProvider:
    """Create the configured provider without fetching any secret values."""
    provider_name = os.getenv("SECRET_PROVIDER", "local").strip().lower()

    if provider_name == "local":
        return LocalMockSecretProvider(
            {
                "payment-api/example": "local-dummy-secret",
            }
        )

    if provider_name == "aws":
        region_name = os.getenv("AWS_REGION", "").strip()
        if not region_name:
            raise ValueError(
                "AWS_REGION must be set when SECRET_PROVIDER=aws"
            )

        return AwsSecretsManagerProvider(region_name=region_name)

    raise ValueError(
        "Unsupported SECRET_PROVIDER. Choose 'local' or 'aws'."
    )

==> applications/payment-api/app/secrets.py <==
"""Secret-provider interface and implementations for SecureOps.

LocalMockSecretProvider is for local development and tests only.
It must never be used to store real credentials.
"""

from typing import Protocol

import boto3
from botocore.exceptions import BotoCoreError, ClientError


class SecretProvider(Protocol):
    """Interface implemented by local and cloud secret providers."""

    def get_secret(self, name: str) -> str:
        """Return a secret value by name."""
        ...


class LocalMockSecretProvider:
    """Return explicitly supplied dummy values for local development."""

    def __init__(self, values: dict[str, str] | None = None) -> None:
        self._values = values or {}

    def get_secret(self, name: str) -> str:
        """Return a configured dummy value or fail without exposing values."""
        value = self._values.get(name)
        if value is None:
            raise KeyError(f"Secret not configured: {name}")
        return value


class AwsSecretsManagerProvider:
    """Retrieve secret strings from AWS Secrets Manager."""

    def __init__(self, region_name: str, client=None) -> None:
        if not region_name:
            raise ValueError("AWS region_name must not be empty")

        self._client = (
            client
            if client is not None
            else boto3.client("secretsmanager", region_name=region_name)
        )

    def get_secret(self, name: str) -> str:
        """Return SecretString  without logging the secret value."""
        if not name:
            raise ValueError("Secret name must not be empty")

        try:
            response = self._client.get_secret_value(SecretId=name)
        except (BotoCoreError, ClientError) as exc:
            raise RuntimeError(
                f"Unable to retrieve secret: {name}"
            ) from exc

        value = response.get("SecretString")
        if not isinstance(value, str):
            raise ValueError(
                f"Secret does not contain a string value: {name}"
            )

        return value

    rm -f "$TEMP_MANIFEST"

log() {
    printf '\n[%s] %s\n' "$1" "$2"
}

fail() {
    printf 'ERROR: %s\n' "$1" >&2
    exit 1
}

log INFO "SecureOps repository validation started"

for tool in bash kubectl; do
    command -v "$tool" >/dev/null 2>&1 ||
        fail "Required command not found: $tool"
done

log CHECK "Checking Bash syntax"
mapfile -t SHELL_SCRIPTS < <(
    find "$REPO_ROOT/scripts" -type f -name '*.sh' -print | sort
)

((${#SHELL_SCRIPTS[@]} > 0)) ||
    fail "No shell scripts found under scripts/"

for script in "${SHELL_SCRIPTS[@]}"; do
    bash -n "$script"
    printf 'PASS: %s\n' "${script#"$REPO_ROOT"/}"
done

if command -v shellcheck >/dev/null 2>&1; then
    log CHECK "Running ShellCheck"
    shellcheck "${SHELL_SCRIPTS[@]}"
else
    log WARN "ShellCheck not installed; static shell analysis skipped"
fi

log CHECK "Running Payment API tests and lint"
"$REPO_ROOT/scripts/quality-gate.sh"

log CHECK "Rendering Kubernetes base manifests"
kubectl kustomize "$REPO_ROOT/kubernetes/base" > "$TEMP_MANIFEST"

grep -q '^kind: Deployment$' "$TEMP_MANIFEST" ||
    fail "Rendered manifests do not contain a Deployment"

#!/usr/bin/env bash
set -Eeuo pipefail

KIND_NODE="${KIND_NODE:-devops-control-plane}"
NAMESPACE="${NAMESPACE:-default}"

if ! docker inspect "$KIND_NODE" >/dev/null 2>&1; then
  echo "ERROR: Kind node container '$KIND_NODE' not found." >&2
  exit 1
fi

kube() {
  docker exec "$KIND_NODE" kubectl \
    --kubeconfig /etc/kubernetes/admin.conf \
    --namespace "$NAMESPACE" "$@"
}

echo "==> Checking deployment rollout"
kube rollout status deployment/payment-api --timeout=120s

echo "==> Checking replicas"
desired="$(kube get deployment payment-api -o jsonpath='{.spec.replicas}')"
ready="$(kube get deployment payment-api -o jsonpath='{.status.readyReplicas}')"
ready="${ready:-0}"

echo "Desired: $desired | Ready: $ready"

if [[ "$ready" != "$desired" || "$desired" -lt 1 ]]; then
  echo "ERROR: deployment replicas are not all ready." >&2
  exit 1
fi

echo "==> Checking Service EndpointSlices"
endpoints="$(kube get endpointslice \
  -l kubernetes.io/service-name=payment-api \
  -o jsonpath='{range .items[*].endpoints[?(@.conditions.ready==true)]}{.addresses[0]}{" "}{end}')"

if [[ -z "$endpoints" ]]; then
  echo "ERROR: payment-api Service has no ready EndpointSlice endpoints." >&2
  exit 1
fi

echo "Ready endpoints: $endpoints"

echo "==> Testing application endpoints from a payment-api pod"
pod="$(kube get pods -l app=payment-api \
  -o jsonpath='{.items[0].metadata.name}')"

if [[ -z "$pod" ]]; then
  echo "ERROR: no payment-api pod found." >&2
  exit 1
fi

kube exec "$pod" -- python -c '
import json
import urllib.request

base = "http://payment-api:8000"
checks = {
    "/": lambda b: json.loads(b).get("service") == "payment-api",
    "/health": lambda b: json.loads(b).get("status") == "healthy",
    "/ready": lambda b: json.loads(b).get("status") == "ready",
    "/metrics/": lambda b: "python_" in b or "process_" in b,
}

for path, validate in checks.items():
    with urllib.request.urlopen(base + path, timeout=5) as response:
        body = response.read().decode()
        if response.status != 200 or not validate(body):
            raise SystemExit("FAIL " + path + ": unexpected response")
        print("PASS", path, "HTTP", response.status)

print("All application endpoint checks passed.")
'

echo "SUCCESS: payment-api verification completed."
